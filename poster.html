نمونه کوئری‌ها
پیش‌نیازهای هر درس
SELECT c1.CourseName AS درس_اصلی, c2.CourseName AS پیش_نیاز FROM Prerequisite p JOIN Courses c1 ON p.CourseID = c1.CourseID JOIN Courses c2 ON p.PrerequisiteID = c2.CourseID;
دانشجویان با بیشترین واحد
SELECT s.FirstName, s.LastName, SUM(c.Credits) AS مجموع_واحد FROM Enrollment e JOIN Students s ON e.StudentID = s.StudentID JOIN Section sec ON e.SectionID = sec.SectionID JOIN Courses c ON sec.CourseID = c.CourseID WHERE e.Status = 'Registered' GROUP BY s.StudentID ORDER BY مجموع_واحد DESC LIMIT 5;
